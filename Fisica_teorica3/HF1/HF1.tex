\section*{Ecuaciones de Maxwell y Cv}
$
\begin{aligned}
    +\bigg(\frac{\partial T}{\partial V}\bigg)_S &= -\bigg(\frac{\partial p}{\partial S}\bigg)_V = \frac{\partial^2 U}{\partial S \partial V} \\
    +\bigg(\frac{\partial T}{\partial p}\bigg)_S &= +\bigg(\frac{\partial V}{\partial S}\bigg)_p = \frac{\partial^2 H}{\partial S \partial p} \\
    +\bigg(\frac{\partial S}{\partial V}\bigg)_T &= +\bigg(\frac{\partial p}{\partial T}\bigg)_V = \frac{\partial^2 F}{\partial T \partial V} \\
    -\bigg(\frac{\partial S}{\partial p}\bigg)_T &= +\bigg(\frac{\partial V}{\partial T}\bigg)_p = \frac{\partial^2 G}{\partial T \partial p} \\

\end{aligned}
$

\subsection{Bayes}
\begin{equation*}
    P(A \cap B) = P(A | B)P(B) = P(B|A)P(A)
\end{equation*}

\begin{equation*}
    P(A | B) = \frac{P(B | A)P(A)}{P(B)}
\end{equation*}
